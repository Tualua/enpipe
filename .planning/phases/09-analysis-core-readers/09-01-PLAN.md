STATE_ENABLED = "enabled"; STATE_UNAVAILABLE = "unavailable"
STATE_REJECTED = "rejected"; STATE_NOT_REQUESTED = "not_requested"
ROW_OK = "ok"; ROW_ERROR = "error"

@dataclass(frozen=True)
class ChromaDetail:
    status: str                         # CHROMA_OK | CHROMA_NO_DETAILS
    n_blocks: int
    median: Optional[float] = None
    p10: Optional[float] = None
    ratio: Optional[float] = None       # Σ(s·e)/Σ(s·s) по всем блокам (D-02)
    worst_value: Optional[float] = None
    worst_x: Optional[int] = None       # пиксели Y (D-06)
    worst_y: Optional[int] = None

@dataclass(frozen=True)
class LumaTile:   x: int; y: int; w: int; h: int; psnr: float
@dataclass(frozen=True)
class ChromaTile: x: int; y: int; w: int; h: int; psnr_u: float; psnr_v: float; mse_uv: float
@dataclass(frozen=True)
class DetailTile: x: int; y: int; w: int; h: int; detail: float

@dataclass(frozen=True)
class FrameMetrics:
    psnr_y: float; psnr_u: float; psnr_v: float          # inf допустим
    worst_tile_y: LumaTile
    worst_tile_c: ChromaTile
    nits_mean: float
    shadow_fraction: float
    detail: Optional[float]                               # None: источник плоский
    worst_tile_detail: Optional[DetailTile]
    delta_e_itp_mean: float
    delta_e_itp_p99: float
    chroma: ChromaDetail

@dataclass(frozen=True)
class SourceStatus:
    source: str                         # SOURCE_*
    state: str                          # STATE_*
    reason: Optional[str] = None        # русский текст причины отказа/недоступности
    notes: Tuple[str, ...] = ()         # предупреждения (не отказ)
    count: int = 0                      # сколько кандидатов дал источник

@dataclass(frozen=True)
class RawCandidate:
    frame: int; source: str; rank: float; tags: Tuple[str, ...] = ()
    # rank — ключ порядка ВНУТРИ источника (меньше = важнее): vmaf — значение VMAF,
    # csv — лучшая позиция в топе (0-based), manual — 0.0; между источниками не сравнивается

@dataclass(frozen=True)
class Candidate:
    frame: int; primary: str; sources: Tuple[str, ...]; tags: Tuple[str, ...] = (); scene: Optional[int] = None

@dataclass(frozen=True)
class ImageRef:
    kind: str; path: str; caption: str

@dataclass(frozen=True)
class FrameRow:
    frame: int
    status: str                         # ROW_OK | ROW_ERROR (RPT-04 в фазе 11)
    sources: Tuple[str, ...] = ()
    scene: Optional[int] = None
    scene_pos: Optional[int] = None
    time_sec: Optional[float] = None
    vmaf: Optional[float] = None
    metrics: Optional[FrameMetrics] = None
    error: Optional[str] = None
    alignment: Optional[str] = None     # фаза 10
    alignment_shift: Optional[int] = None
    flags: Tuple[str, ...] = ()         # фаза 11 (CND-06)
    severity: Optional[float] = None    # фаза 11 (RPT-03)
    images: Tuple[ImageRef, ...] = ()   # фаза 11 (RPT-02)

@dataclass(frozen=True)
class RunMeta:
    src_name: str; enc_name: str
    tool: str = "enpipe analyze"
    created_at: Optional[str] = None
    enpipe_version: Optional[str] = None
    ffmpeg_version: Optional[str] = None
    qsvencc_version: Optional[str] = None
    width: Optional[int] = None; height: Optional[int] = None
    fps: Optional[float] = None; frame_count: Optional[int] = None

@dataclass(frozen=True)
class Threshold:
    name: str; value: Union[int, float]; provisional: bool = True

@dataclass(frozen=True)
class PercentileSummary:
    metric: str; n: int
    minimum: Optional[float] = None; p1: Optional[float] = None; p5: Optional[float] = None
    p10: Optional[float] = None; median: Optional[float] = None

@dataclass(frozen=True)
class AnalysisReport:
    meta: RunMeta

    sources: Tuple[SourceStatus, ...]
    candidates: Tuple[Candidate, ...]
    frames: Tuple[FrameRow, ...]
    summary: Tuple[PercentileSummary, ...]
